// 🤖 UltraAI Voice System const SpeechRecognition = window.SpeechRecognition || window.webkitSpeechRecognition; let recognition; let voiceMode = false; if (SpeechRecognition) { recognition = new SpeechRecognition(); recognition.continuous = false; recognition.interimResults = false; // Normal Mic function startListening() { setLanguage(); recognition.start(); } // 📞 Voice Mode Start function startVoiceMode() { voiceMode = true; speakText("नमस्ते, मैं UltraAI हूँ। बताइए मैं आपकी क्या मदद कर सकता हूँ।"); setTimeout(() => { recognition.start(); }, 2000); } function setLanguage(){ let lang = document.getElementById("languageSelect"); if(lang){ recognition.lang = lang.value; }else{ recognition.lang = "hi-IN"; } } // User Voice Result recognition.onresult = function(event){ let text = event.results[0][0].transcript; document.getElementById("userInput").value = text; sendMessage(); }; // AI बोलने के बाद फिर सुनना recognition.onend = function(){ if(voiceMode){ setTimeout(()=>{ recognition.start(); },1500); } }; recognition.onerror = function(error){ console.log(error); if(voiceMode){ setTimeout(()=>{ recognition.start(); },2000); } }; } // 🔊 AI Voice function speakText(text){ let speech = new SpeechSynthesisUtterance(text); let lang = document.getElementById("languageSelect"); if(lang){ speech.lang = lang.value; }else{ speech.lang="hi-IN"; } speech.rate = 1; speech.pitch = 1.2; let voices = window.speechSynthesis.getVoices(); let female = voices.find(v => v.name.toLowerCase().includes("female") || v.name.toLowerCase().includes("zira") || v.name.toLowerCase().includes("google") ); if(female){ speech.voice = female; } window.speechSynthesis.speak(speech); } // Voice Mode बंद करने के लिए function stopVoiceMode(){ voiceMode = false; window.speechSynthesis.cancel(); if(recognition){ recognition.stop(); } }
